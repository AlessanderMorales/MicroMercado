================================================================================
COMMON STEPS - STEPS COMPARTIDOS ENTRE TODOS LOS FEATURES
================================================================================

DESCRIPCIÓN GENERAL:
La clase CommonSteps contiene step definitions que son compartidos y reutilizados
por todos los features del proyecto (Categorías, Clientes, Productos y Ventas).
Estos steps implementan validaciones genéricas y operaciones comunes que se repiten
en múltiples escenarios.

UBICACIÓN: Pruebas BBD/StepDefinitions/CommonSteps.cs

--------------------------------------------------------------------------------
PROPÓSITO Y ARQUITECTURA
--------------------------------------------------------------------------------

Los Common Steps implementan el principio DRY (Don't Repeat Yourself) en BDD,
permitiendo:

1. REUTILIZACIÓN DE CÓDIGO:
   - Evitar duplicación de validaciones idénticas
   - Mantener consistencia en las verificaciones
   - Reducir mantenimiento (un solo lugar para cambios)

2. COMPARTIR CONTEXTO:
   - Usar ScenarioContext para pasar datos entre steps
   - Validar resultados almacenados por otros step definitions
   - Mantener estado del escenario actual

3. VALIDACIONES GENÉRICAS:
   - Verificar éxito o fallo de operaciones CRUD
   - Validar estados de creación, actualización y eliminación
   - Comprobar excepciones y errores

--------------------------------------------------------------------------------
SCENARIOCONTEXT: EL CONTENEDOR COMPARTIDO
--------------------------------------------------------------------------------

¿QUÉ ES ScenarioContext?

ScenarioContext es un diccionario proporcionado por Reqnroll que:

1. ALMACENA DATOS DURANTE UN ESCENARIO:
   - Vive durante todo el ciclo de vida de un escenario
   - Permite compartir información entre diferentes steps
   - Se limpia automáticamente al finalizar el escenario

2. ESTRUCTURA CLAVE-VALOR:
   
   // Almacenar
   _scenarioContext["CreationResult"] = categoryDTO;
   _scenarioContext["UpdateException"] = exception;
   _scenarioContext["ClientId"] = client.Id;
   
   // Recuperar
   var result = _scenarioContext["CreationResult"];
   var clientId = (int)_scenarioContext["ClientId"];
   
   // Verificar existencia
   if (_scenarioContext.ContainsKey("UpdateResult"))
   {
       // Hacer algo con el resultado
   }

3. INYECCIÓN DE DEPENDENCIAS:
   
   Reqnroll inyecta automáticamente el ScenarioContext en los constructores
   de las clases de step definitions:
   
   public CategoryIntegrationSteps(ScenarioContext scenarioContext)
   {
       _scenarioContext = scenarioContext;  // ? Inyectado por Reqnroll
   }

FLUJO DE DATOS TÍPICO:

Escenario: Actualizar cliente
?
?? [Given] existe un cliente "Juan" con TaxDocument "123"
?   ?  ? ClientIntegrationSteps crea cliente en BD
?   ?? Guarda: _scenarioContext["ClientId"] = client.Id
?
?? [When] actualizo el cliente con nuevos datos
?   ?  ? ClientIntegrationSteps recupera: clientId = _scenarioContext["ClientId"]
?   ?  ? Actualiza cliente en BD
?   ?? Guarda: _scenarioContext["UpdateResult"] = updatedClient
?
?? [Then] la actualizacion debe ser exitosa
    ?  ? CommonSteps verifica: _scenarioContext["UpdateResult"] != null
    ?? Assert.NotNull(_scenarioContext["UpdateResult"])

CLAVES COMUNES USADAS EN EL PROYECTO:

| Clave                | Tipo          | Usado en Features    | Propósito                        |
|---------------------|---------------|---------------------|----------------------------------|
| "CreationResult"    | DTO / null    | Todos (Create)      | Resultado de operación CREATE    |
| "UpdateResult"      | DTO / null    | Todos (Update)      | Resultado de operación UPDATE    |

ESCENARIO: CA-01 - Crear categoría básica (Happy Path 1)
--------------------------------------------------------------------------------

DESCRIPCIÓN:
Verifica que se puede crear una categoría con datos básicos correctos.
La categoría debe crearse con Status activo (1) automáticamente.

IMPLEMENTACIÓN:
- Step Definition: CategoryIntegrationSteps.cs
- Método principal: WhenCreoUnaCategoria(Table table)

CÓDIGO PASO A PASO:

[When(@"creo una categoria con los siguientes datos:")]
public async Task WhenCreoUnaCategoria(Table table)
{
    try
    {
        string name = string.Empty;
        string description = string.Empty;

        // 1. Extraer datos de la tabla del escenario
        foreach (var row in table.Rows)
        {
            var campo = row["Campo"];
            var valor = row["Valor"];

            switch (campo)
            {
                case "Name":
                    name = valor;
                    break;
                case "Description":
                    description = valor;
                    break;
            }
        }

        // 2. Crear el DTO con los datos extraídos
        var dto = new CreateCategoryDTO
        {
            Name = name,
            Description = description
        };

        // 3. Llamar al servicio para crear la categoría
        _resultCategory = await _categoryService.CreateCategoryAsync(dto);
        
        // 4. Guardar resultado en el contexto del escenario
        _scenarioContext["CreationResult"] = _resultCategory;
    }
    catch (Exception ex)
    {
        _exception = ex;
        _scenarioContext["CreationResult"] = null;
    }
}

VALIDACIONES (THEN):

[Then(@"la categoria debe crearse exitosamente")]
public void ThenLaCategoriDebeCrearseExitosamente()
{
    Assert.NotNull(_resultCategory);  // Verifica que se creó algo
    Assert.Null(_exception);          // Verifica que no hubo excepciones
}

[Then(@"el nombre debe ser ""(.*)""")]
public void ThenElNombreDebeSer(string expectedName)
{
    Assert.Equal(expectedName, _resultCategory?.Name);
}

[Then(@"el Status debe ser (.*)")]
public void ThenElStatusDebeSer(int expectedStatus)
{
    Assert.Equal((byte)expectedStatus, _resultCategory?.Status);
}

ESCENARIO: CA-04 - Actualizar categoría exitosamente (Happy Path 1)
--------------------------------------------------------------------------------

DESCRIPCIÓN:
Verifica que se puede actualizar una categoría existente cambiando
tanto el nombre como la descripción.

IMPLEMENTACIÓN:

PASO 1 - PREPARACIÓN (GIVEN):
[Given(@"existe una categoria con nombre ""(.*)""")]
public async Task GivenExisteUnaCategoriaConNombre(string nombre)
{
    _context.ChangeTracker.Clear();  // Limpiar tracking de EF

    // Crear categoría de prueba en la BD
    var category = new Category
    {
        Id = _nextCategoryId++,
        Name = nombre,
        Description = "Descripción de prueba",
        Status = 1,
        LastUpdate = DateTime.Now
    };

    _context.Categories.Add(category);
    await _context.SaveChangesAsync();
    _context.ChangeTracker.Clear();

    // Guardar ID para uso posterior
    _scenarioContext[$"Category_{nombre}"] = category.Id;
    _scenarioContext["LastCategoryId"] = category.Id;
}

PASO 2 - ACTUALIZACIÓN (WHEN):
[When(@"actualizo la categoria con:")]
public async Task WhenActualizoLaCategoria(Table table)
{
    try
    {
        _context.ChangeTracker.Clear();

        // 1. Recuperar ID del contexto del escenario
        byte categoryId = (byte)_scenarioContext["LastCategoryId"];

        // 2. Obtener categoría existente de la BD
        var existingCategory = await _context.Categories
            .AsNoTracking()
            .FirstOrDefaultAsync(c => c.Id == categoryId);

        if (existingCategory == null)
            throw new InvalidOperationException($"Category ID {categoryId} not found");

        // 3. Inicializar con valores actuales
        string name = existingCategory.Name;
        string description = existingCategory.Description ?? string.Empty;
        byte status = existingCategory.Status;

        // 4. Aplicar cambios de la tabla del escenario
        foreach (var row in table.Rows)
        {
            var campo = row["Campo"];
            var valor = row["Valor"];

            switch (campo)
            {
                case "Name":
                    name = valor;
                    break;
                case "Description":
                    description = valor;
                    break;
                case "Status":
                    status = byte.Parse(valor);
                    break;
            }
        }

        // 5. Crear DTO con datos actualizados
        var dto = new UpdateCategoryDTO
        {
            Id = categoryId,
            Name = name,
            Description = description,
            Status = status
        };

        // 6. Llamar al servicio
        _resultCategory = await _categoryService.UpdateCategoryAsync(dto);
        _scenarioContext["UpdateResult"] = _resultCategory;
        _scenarioContext["UpdateException"] = null;
    }
    catch (Exception ex)
    {
        _exception = ex;
        _scenarioContext["UpdateResult"] = null;
        _scenarioContext["UpdateException"] = ex;
    }
}

PASO 3 - VALIDACIÓN (THEN):
[Then(@"los datos deben reflejarse en la base de datos")]
public async Task ThenLosDatosDebenReflejarseEnLaBaseDeDatos()
{
    Assert.NotNull(_resultCategory);
    _context.ChangeTracker.Clear();
    
    // Verificar en BD que los cambios se guardaron
    var dbCategory = await _context.Categories
        .AsNoTracking()
        .FirstOrDefaultAsync(c => c.Id == _resultCategory.Id);
    
    Assert.NotNull(dbCategory);
    Assert.Equal(_resultCategory.Name, dbCategory.Name);
}

[Given(@"tengo los siguientes productos en el carrito:")]

    _saleDto.Items = new List<SaleDTO.SaleItemDTO>();

    {

        var quantity = short.Parse(row["Quantity"]);

        // Buscar ID del producto en el diccionario

        {
            var product = _context.Products.Find(productId);

            // Agregar item al carrito
            _saleDto.Items.Add(new SaleDTO.SaleItemDTO
            {
                ProductId = productId,
                Quantity = quantity,
                Price = product!.Price
            });
        }
    }
}

PASO 2 - PROCESAR LA VENTA:

[When(@"proceso la venta con los siguientes datos:")]
public async Task WhenProcesoLaVentaConLosSiguientesDatos(Table table)
{
    // 1. Obtener ClientId del contexto
    _saleDto.ClientId = (int)_scenarioContext["ClientId"];
    
    // 2. Procesar datos adicionales de la tabla
    foreach (var row in table.Rows)
    {
        var campo = row["Campo"].Trim();
        var valor = row["Valor"].Trim();
        
        if (campo.Equals("PaymentType", StringComparison.OrdinalIgnoreCase))
        {
            _saleDto.PaymentType = valor.Equals("Efectivo", StringComparison.OrdinalIgnoreCase) 
                ? (byte)1 
                : (byte)2;
        }
        else if (campo.Equals("CashReceived", StringComparison.OrdinalIgnoreCase))
        {
            _saleDto.CashReceived = decimal.Parse(valor);
        }
    }

    // 3. Calcular totales
    _saleDto.TotalAmount = _saleDto.Items.Sum(i => i.Quantity * i.Price);
    _saleDto.Change = _saleDto.CashReceived - _saleDto.TotalAmount;

    // 4. Llamar al servicio para crear la venta
    _saleResult = await _saleService.CreateSaleAsync(_saleDto);
}
